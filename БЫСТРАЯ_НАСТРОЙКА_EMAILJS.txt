║          БЫСТРАЯ НАСТРОЙКА EMAILJS - 5 ШАГОВ                ║
╚══════════════════════════════════════════════════════════════╝

📌 После выполнения этих шагов заявки будут автоматически
   отправляться на pendeho098rus@yandex.ru

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1️⃣  РЕГИСТРАЦИЯ (2 мин)
   → Откройте: https://www.emailjs.com/
   → Зарегистрируйтесь (можно через Google)
   → Подтвердите email если нужно

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

2️⃣  ПОДКЛЮЧЕНИЕ EMAIL (5 мин)
   → EmailJS → "Email Services" → "Add New Service"
   → Выберите "Gmail" (рекомендуется) или "Custom SMTP"
   → Подключите ваш email аккаунт
   → ✅ СКОПИРУЙТЕ Service ID (например: service_abc123)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

3️⃣  СОЗДАНИЕ ШАБЛОНА (3 мин)
   → EmailJS → "Email Templates" → "Create New Template"
   → Заполните:
     • Template Name: Заявка на съемку
     • To Email: {{to_email}}
     • Subject: {{subject}}
     • Content: {{message}}
     • From Name: Заявки на съемку
     • From Email: ваш email
   → Нажмите "Save"
   → ✅ СКОПИРУЙТЕ Template ID (например: template_xyz789)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

4️⃣  ПОЛУЧЕНИЕ КЛЮЧА (1 мин)
   → EmailJS → "Account" → "General"
   → Найдите "Public Key" в разделе "API Keys"
   → ✅ СКОПИРУЙТЕ Public Key (например: abcdefghijklmnop)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

5️⃣  ЗАПОЛНЕНИЕ КОДА (2 мин)
   → Откройте файл: email-service.js
   → Найдите строки 22-25 (около this.emailjsConfig)
   → Замените:
     • 'YOUR_SERVICE_ID'   → ваш Service ID из шага 2
     • 'YOUR_TEMPLATE_ID'  → ваш Template ID из шага 3
     • 'YOUR_PUBLIC_KEY'   → ваш Public Key из шага 4
   → Сохраните файл (Ctrl+S)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✅  ПРОВЕРКА (1 мин)
   → Откройте index.html в браузере
   → Войдите в систему
   → Создайте тестовую заявку
   → Нажмите "Отправить заявку"
   → Должно появиться: "Заявка успешно отправлена на email!"
